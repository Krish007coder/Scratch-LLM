1. Under the Hood: Routing & CORS Preflight

To enable cross-origin requests (CORS) from a separate frontend application, we configure Flask to return permissive CORS headers. When browser applications make cross-origin writes (e.g. POST or DELETE requests), they first send an OPTIONS preflight request. The server must handle these preflight requests by returning a 204 No Content response with valid CORS headers before the actual write request is executed.

2. Request Validation and Serialization

Production API interfaces must validate incoming payloads and return structured error responses (e.g. 400 Bad Request) for malformed data. Additionally, coordinate values returned in search responses are rounded to 4 decimal places before serialization, reducing the size of the JSON payloads transferred over the network.

pythonimport os
from flask import Flask, jsonify, request, send_file
from vectordb import DocumentDB, OllamaClient, VectorDB, get_dist_fn

DIMS = 16

def _cors(response):
    response.headers["Access-Control-Allow-Origin"] = "*"
    response.headers["Access-Control-Allow-Methods"] = "GET, POST, DELETE, OPTIONS"
    response.headers["Access-Control-Allow-Headers"] = "Content-Type"
    return response

def _parse_vec(s: str):
    res = []
    for token in s.split(","):
        token = token.strip()
        try: res.append(float(token))
        except ValueError: pass
    return res

def create_app(db: VectorDB, doc_db: DocumentDB, ollama: OllamaClient) -> Flask:
    app = Flask(__name__, static_folder=None)

    @app.before_request
    def handle_preflight():
        if request.method == "OPTIONS":
            from flask import make_response
            resp = make_response("", 204)
            _cors(resp)
            return resp

    @app.after_request
    def attach_cors(response):
        return _cors(response)

    @app.route("/")
    def index():
        html_path = os.path.join(os.path.dirname(__file__), "index.html")
        return send_file(html_path, mimetype="text/html")

    @app.route("/search")
    def search():
        q = _parse_vec(request.args.get("v", ""))
        if len(q) != DIMS: return jsonify({{"error": f"need {{DIMS}}D vector"}}), 400
        k = int(request.args.get("k", 5))
        metric = request.args.get("metric", "cosine")
        algo = request.args.get("algo", "hnsw")
        out = db.search(q, k, metric, algo)
        return jsonify({{
            "results": [{{
                "id": h.id, "metadata": h.meta, "category": h.cat,
                "distance": round(h.dist, 6), "embedding": [round(x, 4) for x in h.emb]
            }} for h in out.hits],
            "latencyUs": out.latency_us, "algo": out.algo, "metric": out.metric
        }})

    @app.route("/doc/ask", methods=["POST"])
    def doc_ask():
        body = request.get_json(silent=True) or {{}}
        question = body.get("question", "").strip()
        k = int(body.get("k", 3))
        if not question: return jsonify({{"error": "need question"}}), 400
        q_emb = ollama.embed(question)
        if not q_emb: return jsonify({{"error": "Ollama unavailable"}}), 503
        hits = doc_db.search(q_emb, k)
        context_parts = [f"[{{i + 1}}] {{doc.title}}:\\n{{doc.text}}" for i, (_, doc) in enumerate(hits)]
        context_str = "\\n\\n".join(context_parts)
        prompt = (
            "You are a helpful assistant. Answer the user's question directly. "
            "Use the provided context if it contains relevant information. "
            "IMPORTANT: Do NOT mention 'context' or 'provided text' in your answer.\\n\\n"
            f"Context:\\n{{context_str}}\\n\\nQuestion: {{question}}\\n\\nAnswer:"
        )
        answer = ollama.generate(prompt)
        return jsonify({{
            "answer": answer,
            "contexts": [{{"id": doc.id, "title": doc.title, "text": doc.text, "distance": round(d, 4)}} for d, doc in hits]
        }})

    return app
Code Analysis:
  • @app.before_request: Intercepts preflight requests and returns early, preventing cross-origin writes from triggering full request handling.
  • silent=True: Prevents the JSON parser from throwing exceptions on malformed requests, allowing the endpoint to handle formatting errors gracefully.
