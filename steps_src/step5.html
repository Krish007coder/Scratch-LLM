2. Brute Force Implementation

vectordb/bruteforce.py
Exhaustive Linear Scan
pythonimport heapq
from typing import List, Tuple
from .types import VectorItem
from .metrics import DistFn

DistId = Tuple[float, int]

class BruteForce:
    def __init__(self) -> None:
        self.items: List[VectorItem] = []

    def insert(self, item: VectorItem) -> None:
        self.items.append(item)

    def knn(self, q: List[float], k: int, dist: DistFn) -> List[DistId]:
        if not self.items: return []
        heap = []
        for item in self.items:
            d = dist(q, item.emb)
            if len(heap) < k:
                heapq.heappush(heap, (-d, item.id))
            elif d < -heap[0][0]:
                heapq.heappushpop(heap, (-d, item.id))
        res = [(-d, nid) for d, nid in heap]
        res.sort(key=lambda x: x[0])
        return res

    def remove(self, item_id: int) -> None:
        self.items = [v for v in self.items if v.id != item_id]

3. KD-Tree Space Partitioning

vectordb/kdtree.py
Binary Space Partitioning Tree
pythonimport heapq
from typing import List, Optional, Tuple
from .types import VectorItem
from .metrics import DistFn

DistId = Tuple[float, int]

class _KDNode:
    __slots__ = ("item", "left", "right")
    def __init__(self, item: VectorItem) -> None:
        self.item = item
        self.left = None
        self.right = None

class KDTree:
    def __init__(self, dims: int) -> None:
        self.dims = dims
        self._root = None

    def insert(self, item: VectorItem) -> None:
        self._root = self._insert(self._root, item, depth=0)

    def _insert(self, node: Optional[_KDNode], item: VectorItem, depth: int) -> _KDNode:
        if node is None: return _KDNode(item)
        axis = depth % self.dims
        if item.emb[axis] < node.item.emb[axis]:
            node.left = self._insert(node.left, item, depth + 1)
        else:
            node.right = self._insert(node.right, item, depth + 1)
        return node

    def rebuild(self, items: List[VectorItem]) -> None:
        self._root = None
        for item in items:
            self.insert(item)

    def knn(self, q: List[float], k: int, dist: DistFn) -> List[DistId]:
        heap = []
        self._knn(self._root, q, k, 0, dist, heap)
        res = [(-d, nid) for d, nid in heap]
        res.sort()
        return res

    def _knn(self, node: Optional[_KDNode], q: List[float], k: int, depth: int, dist: DistFn, heap: List[Tuple[float, int]]) -> None:
        if node is None: return
        dn = dist(q, node.item.emb)
        if len(heap) < k or dn < -heap[0][0]:
            heapq.heappush(heap, (-dn, node.item.id))
            if len(heap) > k: heapq.heappop(heap)
        axis = depth % self.dims
        diff = q[axis] - node.item.emb[axis]
        closer = node.left if diff < 0 else node.right
        farther = node.right if diff < 0 else node.left
        self._knn(closer, q, k, depth + 1, dist, heap)
        if len(heap) < k or abs(diff) < -heap[0][0]:
            self._knn(farther, q, k, depth + 1, dist, heap)

4. HNSW Layered Graph

vectordb/hnsw.py
Hierarchical Navigable Small World Graph
pythonimport math
import random
import heapq
from typing import Dict, List, Optional, Tuple, Set
from .types import VectorItem
from .metrics import DistFn

DistId = Tuple[float, int]

class _Node:
    __slots__ = ("item", "max_lyr", "nbrs")
    def __init__(self, item: VectorItem, max_lyr: int) -> None:
        self.item = item
        self.max_lyr = max_lyr
        self.nbrs: List[List[int]] = [[] for _ in range(max_lyr + 1)]

class HNSW:
    def __init__(self, M: int = 16, ef_build: int = 200, seed: int = 42) -> None:
        self.M = M
        self.M0 = 2 * M
        self.ef_build = ef_build
        self.mL = 1.0 / math.log(M)
        self._rng = random.Random(seed)
        self._graph: Dict[int, _Node] = {}
        self._top_layer: int = -1
        self._entry_pt: Optional[int] = None

    def _rand_level(self) -> int:
        return int(math.floor(-math.log(self._rng.random()) * self.mL))

    def _search_layer(self, q: List[float], ep: int, ef: int, lyr: int, dist: DistFn) -> List[DistId]:
        visited: Set[int] = {{ep}}
        d0 = dist(q, self._graph[ep].item.emb)
        cands: List[DistId] = [(d0, ep)]
        found: List[Tuple[float, int]] = [(-d0, ep)]

        while cands:
            cd, cid = heapq.heappop(cands)
            if len(found) >= ef and cd > -found[0][0]: break
            node = self._graph.get(cid)
            if node is None or lyr >= len(node.nbrs): continue
            for nid in node.nbrs[lyr]:
                if nid in visited or nid not in self._graph: continue
                visited.add(nid)
                nd = dist(q, self._graph[nid].item.emb)
                if len(found) < ef or nd < -found[0][0]:
                    heapq.heappush(cands, (nd, nid))
                    heapq.heappush(found, (-nd, nid))
                    if len(found) > ef: heapq.heappop(found)
        res = [(-d, nid) for d, nid in found]
        res.sort()
        return res

    def insert(self, item: VectorItem, dist: DistFn) -> None:
        node_id = item.id
        lvl = self._rand_level()
        node = _Node(item=item, max_lyr=lvl)
        self._graph[node_id] = node

        if self._entry_pt is None:
            self._entry_pt = node_id
            self._top_layer = lvl
            return

        ep = self._entry_pt
        for lc in range(self._top_layer, lvl, -1):
            W = self._search_layer(item.emb, ep, 1, lc, dist)
            if W: ep = W[0][1]

        for lc in range(min(self._top_layer, lvl), -1, -1):
            W = self._search_layer(item.emb, ep, self.ef_build, lc, dist)
            max_m = self.M0 if lc == 0 else self.M
            sel = [nid for _, nid in W[:max_m]]
            self._graph[node_id].nbrs[lc] = sel

            for nid in sel:
                if nid not in self._graph: continue
                nbr_node = self._graph[nid]
                while len(nbr_node.nbrs) <= lc:
                    nbr_node.nbrs.append([])
                conn = nbr_node.nbrs[lc]
                conn.append(node_id)
                if len(conn) > max_m:
                    ds = [(dist(nbr_node.item.emb, self._graph[c].item.emb), c) for c in conn if c in self._graph]
                    ds.sort()
                    nbr_node.nbrs[lc] = [c for _, c in ds[:max_m]]
            if W: ep = W[0][1]

        if lvl > self._top_layer:
            self._top_layer = lvl
            self._entry_pt = node_id

    def knn(self, q: List[float], k: int, ef: int, dist: DistFn) -> List[DistId]:
        if self._entry_pt is None: return []
        ep = self._entry_pt
        for lc in range(self._top_layer, 0, -1):
            W = self._search_layer(q, ep, 1, lc, dist)
            if W: ep = W[0][1]
        return self._search_layer(q, ep, max(ef, k), 0, dist)[:k]

    def remove(self, item_id: int) -> None:
        if item_id not in self._graph: return
        for node in self._graph.values():
            for layer in node.nbrs:
                if item_id in layer: layer.remove(item_id)
        if self._entry_pt == item_id:
            self._entry_pt = next((nid for nid in self._graph if nid != item_id), None)
        del self._graph[item_id]
