Database Managers

Detailed line-by-line inspection of vectordb.py and documentdb.py

1. Under the Hood: Thread Safety and Locks

In multi-threaded web applications like Flask, multiple worker threads execute database operations concurrently. To protect shared data structures from write corruption (e.g. key overwrites or index state mismatches), we use Python's threading.Lock. When a thread calls a write operation, it acquires the lock, serializing write access and preventing race conditions.

2. Dynamic Dimension Ingestion

Our DocumentDB determines embedding dimensions dynamically based on the first inserted vector. This allows the database wrapper to remain model-agnostic, supporting 384D, 768D, or 1536D layouts without modification. Additionally, DocumentDB switches dynamically between search strategies: using a fast BruteForce index for small collections (< 10 documents) to avoid HNSW entry point graph construction latency, and switching to the HNSW index for larger datasets.

3. Static Index Rebuilding

KD-Trees partition space based on median coordinates along cycling dimensions. Deleting an internal node invalidates these spatial boundaries, causing subsequent branches to become unbalanced. Therefore, our KD-Tree is rebuilt from scratch when an item is deleted. While rebuilding is cheap for small demo datasets, it scales at O(N log N), creating a performance bottleneck for large datasets. In production, deletion is handled via tombstoning, and rebuilds are scheduled as background tasks.

vectordb/vectordb.py
Low-Dimensional Database Manager (16D Demo)
pythonimport threading
import time
from dataclasses import dataclass
from typing import Dict, List, Optional, Tuple

from .bruteforce import BruteForce
from .hnsw import HNSW, GraphInfo
from .kdtree import KDTree
from .metrics import DistFn, get_dist_fn
from .types import VectorItem

@dataclass
class Hit:
    id: int
    meta: str
    cat: str
    emb: List[float]
    dist: float

@dataclass
class SearchOut:
    hits: List[Hit]
    latency_us: int
    algo: str
    metric: str

class VectorDB:
    def __init__(self, dims: int) -> None:
        self.dims = dims
        self._store: Dict[int, VectorItem] = {{}}
        self._bf = BruteForce()
        self._kdt = KDTree(dims)
        self._hnsw = HNSW(M=16, ef_build=200)
        self._lock = threading.Lock()
        self._next_id = 1

    def insert(self, metadata: str, category: str, emb: List[float], dist: DistFn) -> int:
        with self._lock:
            item = VectorItem(id=self._next_id, metadata=metadata, category=category, emb=emb)
            self._next_id += 1
            self._store[item.id] = item
            self._bf.insert(item)
            self._kdt.insert(item)
            self._hnsw.insert(item, dist)
            return item.id

    def remove(self, item_id: int) -> bool:
        with self._lock:
            if item_id not in self._store: return False
            del self._store[item_id]
            self._bf.remove(item_id)
            self._hnsw.remove(item_id)
            self._kdt.rebuild(list(self._store.values()))
            return True

    def search(self, q: List[float], k: int, metric: str, algo: str) -> SearchOut:
        with self._lock:
            dist_fn = get_dist_fn(metric)
            t0 = time.perf_counter()
            if algo == "bruteforce":
                raw = self._bf.knn(q, k, dist_fn)
            elif algo == "kdtree":
                raw = self._kdt.knn(q, k, dist_fn)
            else:
                raw = self._hnsw.knn(q, k, ef=50, dist=dist_fn)
            latency = int((time.perf_counter() - t0) * 1000000)
            hits = [
                Hit(id=nid, meta=self._store[nid].metadata, cat=self._store[nid].category, emb=self._store[nid].emb, dist=d)
                for d, nid in raw if nid in self._store
            ]
            return SearchOut(hits=hits, latency_us=latency, algo=algo, metric=metric)
vectordb/documentdb.py
High-Dimensional Document & RAG Database Manager
pythonimport threading
from typing import Dict, List, Optional, Tuple
from .bruteforce import BruteForce
from .hnsw import HNSW
from .metrics import cosine
from .types import DocItem, VectorItem

class DocumentDB:
    def __init__(self) -> None:
        self._store: Dict[int, DocItem] = {{}}
        self._hnsw = HNSW(M=16, ef_build=200)
        self._bf = BruteForce()
        self._lock = threading.Lock()
        self._next_id = 1
        self._dims = 0

    def insert(self, title: str, text: str, emb: List[float]) -> int:
        with self._lock:
            if self._dims == 0:
                self._dims = len(emb)
            doc = DocItem(id=self._next_id, title=title, text=text, emb=emb)
            self._next_id += 1
            self._store[doc.id] = doc
            vi = VectorItem(id=doc.id, metadata=title, category="doc", emb=emb)
            self._hnsw.insert(vi, cosine)
            self._bf.insert(vi)
            return doc.id

    def remove(self, doc_id: int) -> bool:
        with self._lock:
            if doc_id not in self._store: return False
            del self._store[doc_id]
            self._hnsw.remove(doc_id)
            self._bf.remove(doc_id)
            return True

    def search(self, q: List[float], k: int, max_dist: float = 0.7) -> List[Tuple[float, DocItem]]:
        with self._lock:
            if not self._store: return []
            if len(self._store) < 10:
                raw = self._bf.knn(q, k, cosine)
            else:
                raw = self._hnsw.knn(q, k, ef=50, dist=cosine)
            return [
                (d, self._store[nid])
                for d, nid in raw
                if nid in self._store and d <= max_dist
            ]
Code Analysis:
  • self._lock: Serializes mutations across indices (HNSW, KD-Tree, BruteForce) to prevent threads from corrupting the index states during concurrent writes.
  • if len(self._store) < 10: Switches to a linear scan for small datasets to avoid the overhead of graph construction, optimizing search times for small collections.
